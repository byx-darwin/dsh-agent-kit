# dsh-agent-kit 官网实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在仓库 `website/` 下用 Astro 5 建一个「编辑部长文」风格的中文官网，部署到 `https://byx-darwin.github.io/dsh-agent-kit/`，配置项表由本包源码 schema 生成。

**Architecture:** `website/` 是独立 npm 项目；构建期通过 Vite 直接导入本包 `src/` 下的 schemastery schema 生成配置表；站内链接统一经 `withBasePath`；GitHub Actions 构建并部署到 Pages；vitest 覆盖路径、配置表一致性、GEO 一致性、RSS 与构建产物死链。

**Tech Stack:** Astro 5、@astrojs/mdx 4、@astrojs/sitemap 3、@fontsource（noto-serif-sc、noto-sans-sc、jetbrains-mono）、vitest 4、GitHub Pages（actions/deploy-pages@v4）。

**Spec:** `docs/superpowers/specs/2026-09-24-website-design.md`

## Global Constraints

- 站点：`site: https://byx-darwin.github.io/dsh-agent-kit`，`base: /dsh-agent-kit`；所有站内链接经 `withBasePath(import.meta.env.BASE_URL, path)`。
- 全站中文；不做搜索、深色模式、英文版；对比页不评价第三方产品。
- 配色变量（`global.css`）：`--paper #FAF7F0`、`--card #F1EBDF`、`--ink #2A2622`、`--muted #6B6258`、`--rule #E3DCCD`、`--accent #B4532A`、`--code-bg #1F1B18`、`--code-fg #EFE7DA`；朱红只用于小面积元素。
- 字体：标题 Noto Serif SC 700；正文 Noto Sans SC 400，16px，行高 1.85；代码 JetBrains Mono 13px；全部经 @fontsource 自托管。
- 版式：正文列最宽 720px；首页与架构页可放宽到 1080px；移动端两侧 16px 留白、无横向页面滚动、表格可横向滚动。
- 「壹贰叁肆」依次对应 agentWs、dingtalk、agentTasks、jev。
- 本包未发布到 npm 时，所有安装命令旁显示「尚未发布」提示（`src/data/release.ts` 的 `published: false`）。
- 配置项表由源码 schema 生成，不手抄；版本号取自根 `package.json`。
- `website/` 不进入本包发布内容（根 `package.json` 的 `files` 白名单不变）。
- 仓库设为公开、启用 Pages 必须在用户明确授权后执行，执行前先扫描 git 历史。
- 本地提交已获授权（不 push）；push 与仓库可见性变更需另行确认。

```ts
// website/src/lib/paths.ts
/** 把站内绝对路径拼到 Astro 的部署 base 上。 */
export function withBasePath(base: string, path: string): string {
  const normalizedBase = base === '/' ? '' : base.replace(/\/$/, '')
  const normalizedPath = path.startsWith('/') ? path : `/${path}`
  return `${normalizedBase}${normalizedPath}`
}
```

```ts
// website/src/data/config-tables.ts
import { WsConfig } from '../../../src/ws/config'
import { DingtalkConfig } from '../../../src/dingtalk/config'
import { AgentTasksConfig } from '../../../src/agent-tasks/config'
import { JevConfig } from '../../../src/jev/service'
import { SERVICES, type ServiceId } from './services'

export interface ConfigRow {
  path: string
  type: string
  default: string
  range: string
  description: string
  required: boolean
}

type Node = {
  type: string
  dict?: Record<string, Node>
  list?: Node[]
  inner?: Node
  value?: unknown
  meta?: { default?: unknown; min?: number; max?: number; pattern?: { source: string }; description?: string; required?: boolean }
}

const SCHEMAS: Record<ServiceId, Node> = {
  agentWs: WsConfig as unknown as Node,
  dingtalk: DingtalkConfig as unknown as Node,
  agentTasks: AgentTasksConfig as unknown as Node,
  jev: JevConfig as unknown as Node,
}

function typeLabel(n: Node): string {
  if (n.type === 'union') return n.list!.every((x) => x.type === 'const') ? 'enum' : n.list!.map(typeLabel).join(' | ')
  if (n.type === 'array') return `${typeLabel(n.inner!)}[]`
  if (n.type === 'dict') return `Record<string, ${typeLabel(n.inner!)}>`
  return n.type
}

function formatDefault(n: Node): string {
  const d = n.meta?.default
  if (d === undefined) return '—'
  if (typeof d === 'object') return Array.isArray(d) && d.length === 0 ? '[]' : JSON.stringify(d)
  return String(d)
}

function formatRange(n: Node): string {
  const m = n.meta ?? {}
  if (n.type === 'union' && n.list!.every((x) => x.type === 'const')) return n.list!.map((x) => String(x.value)).join(' | ')
  if (m.min !== undefined && m.max !== undefined) return `[${m.min}, ${m.max}]`
  if (m.min !== undefined) return `≥ ${m.min}`
  if (m.max !== undefined) return `≤ ${m.max}`
  if (m.pattern) return `匹配 /${m.pattern.source}/`
  return '—'
}

function walk(n: Node, prefix: string, out: ConfigRow[], descriptions: Record<string, string>): void {
  if (n.type === 'object' && n.dict) {
    for (const [key, child] of Object.entries(n.dict)) walk(child, prefix ? `${prefix}.${key}` : key, out, descriptions)
    return
  }
  out.push({
    path: prefix,
    type: typeLabel(n),
    default: formatDefault(n),
    range: formatRange(n),
    description: n.meta?.description ?? descriptions[prefix] ?? '',
    required: n.meta?.required === true,
  })
}

export function configRows(id: ServiceId): ConfigRow[] {
  const meta = SERVICES.find((s) => s.id === id)!
  const rows: ConfigRow[] = []
  walk(SCHEMAS[id], '', rows, meta.descriptions)
  return rows
}
```

```ts
// website/src/lib/feed.ts
import { POSTS } from '../data/blog'
import { SITE } from './site'
import { withBasePath } from './paths'

const esc = (s: string) => s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&apos;')

export function renderFeed(site: string, base: string): string {
  const origin = new URL(site).origin
  const link = (path: string) => `${origin}${withBasePath(base, path)}`
  const items = POSTS.map(
    (p) => `    <item>
      <title>${esc(p.title)}</title>
      <link>${link(`/blog/${p.slug}/`)}</link>
      <guid>${link(`/blog/${p.slug}/`)}</guid>
      <pubDate>${new Date(`${p.date}T00:00:00+08:00`).toUTCString()}</pubDate>
      <description>${esc(p.description)}</description>
    </item>`,
  ).join('\n')
  return `<?xml version="1.0" encoding="UTF-8"?>
<rss version="2.0">
  <channel>
    <title>${esc(SITE.name)} 博客</title>
    <link>${link('/blog/')}</link>
    <description>${esc(SITE.positioning)}</description>
    <language>zh-CN</language>
${items}
  </channel>
</rss>
`
}
```

```ts
// website/tests/links.test.ts
import { existsSync, readFileSync, readdirSync, statSync } from 'node:fs'
import { join, relative } from 'node:path'
import { describe, expect, it } from 'vitest'

const DIST = new URL('../dist/', import.meta.url).pathname
const BASE = '/dsh-agent-kit'

function htmlFiles(dir: string): string[] {
  return readdirSync(dir).flatMap((name) => {
    const p = join(dir, name)
    return statSync(p).isDirectory() ? htmlFiles(p) : p.endsWith('.html') ? [p] : []
  })
}

function targetExists(url: string): boolean {
  const clean = decodeURI(url.split('#')[0]!.split('?')[0]!)
  if (!clean.startsWith(`${BASE}/`) && clean !== BASE) return false
  const rel = clean.slice(BASE.length) || '/'
  const p = join(DIST, rel)
  return existsSync(p) && (statSync(p).isFile() || existsSync(join(p, 'index.html')))
}

describe.skipIf(!existsSync(DIST))('built site links', () => {
  it('every internal href/src points to an existing file under the base path', () => {
    const broken: string[] = []
    for (const file of htmlFiles(DIST)) {
      const html = readFileSync(file, 'utf8')
      for (const m of html.matchAll(/\s(?:href|src)="([^"]+)"/g)) {
        const url = m[1]!
        if (/^(https?:|mailto:|#|data:)/.test(url)) continue
        if (!targetExists(url)) broken.push(`${relative(DIST, file)} → ${url}`)
      }
    }
    expect(broken).toEqual([])
  })
})
```
